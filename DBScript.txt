//querying

use wasteManagementDB

db.collectionVehicles.createIndex({ currentLocation: "2dsphere" })

//available vehicles
db.collectionVehicles.find(
  { available: true },
  { _id: 0, vehicleID: 1, registration: 1, vehicleType: 1 }
)

//electric sorted by capacity
db.collectionVehicles.find(
  { fuelType: "Electric" },
  { _id: 0, vehicleID: 1, capacityKg: 1 }
).sort({ capacityKg: -1 })

//no of routes between 20 and 30km
db.collectionRoutes.find({ distanceKm: { $gte: 20, $lte: 30 } }).count()

//organic waste routes or routes over 35km
db.collectionRoutes.find(
  { $or: [ { wasteType: "Organic Waste" }, { distanceKm: { $gt: 35 } } ] },
  { _id: 0, routeID: 1, wasteType: 1, distanceKm: 1 }
)

//routes in tallaght, swords or slucan
db.collectionRoutes.find(
  { cityZone: { $in: ["Tallaght", "Swords", "Lucan"] } },
  { _id: 0, routeID: 1, routeName: 1, cityZone: 1 }
)

//vehicles with a maintenance job over 500
db.collectionVehicles.find(
  { "maintenanceHistory.cost": { $gt: 500 } },
  { _id: 0, vehicleID: 1, maintenanceHistory: 1 }
)

//vehicles with no maintenance before
db.collectionVehicles.find(
  { maintenanceHistory: { $size: 0 } },
  { _id: 0, vehicleID: 1, registration: 1 }
)

//routes with a main st stop of 300kg+
db.collectionRoutes.find(
  { collectionStops: { $elemMatch: { street: "Main Street", weightCollected: { $gte: 300 } } } },
  { _id: 0, routeID: 1, routeName: 1 }
)

//routes with recycling in the name
db.collectionRoutes.find(
  { routeName: { $regex: /recycling/, $options: "i" } },
  { _id: 0, routeID: 1, routeName: 1 }
)

//completed routes from 1 to 7 November
db.collectionRoutes.find(
  {
    collectionDate: { $gte: ISODate("2026-11-01"), $lt: ISODate("2026-11-08") },
    completed: true
  },
  { _id: 0, routeID: 1, collectionDate: 1 }
).sort({ collectionDate: 1 })

//available vehicles within 5km of the city centre
db.collectionVehicles.find({
  currentLocation: {
    $near: {
      $geometry: { type: "Point", coordinates: [-6.2603, 53.3498] },
      $maxDistance: 5000
    }
  },
  available: true
})

//total and average weight by waste type
db.collectionRoutes.aggregate([
  { $group: {
      _id: "$wasteType",
      totalWeight: { $sum: "$totalWeightCollected" },
      avgWeight: { $avg: "$totalWeightCollected" }
  } },
  { $sort: { totalWeight: -1 } }
])

//top 3 vehicles by completed routes
db.collectionRoutes.aggregate([
  { $match: { completed: true } },
  { $group: {
      _id: "$vehicleID",
      routes: { $sum: 1 },
      totalDistance: { $sum: "$distanceKm" }
  } },
  { $sort: { routes: -1, totalDistance: -1 } },
  { $limit: 3 }
])

//tehicles grouped by fuel type
db.collectionVehicles.aggregate([
  { $group: {
      _id: "$fuelType",
      vehicles: { $sum: 1 },
      avgCapacity: { $avg: "$capacityKg" },
      vehicleIDs: { $push: "$vehicleID" }
  } },
  { $sort: { vehicles: -1 } }
])

//general waste totals. largest and smallest collection
db.collectionRoutes.aggregate([
  { $match: { wasteType: "General Waste" } },
  { $group: {
      _id: "$wasteType",
      totalWeight: { $sum: "$totalWeightCollected" },
      maxWeight: { $max: "$totalWeightCollected" },
      minWeight: { $min: "$totalWeightCollected" }
  } }
])

db.createRole({
  role: "adminRole",
  privileges: [
    { resource: { db: "wasteManagementDB", collection: "collectionVehicles" },
      actions: ["insert", "find", "update", "remove"] },
    { resource: { db: "wasteManagementDB", collection: "collectionRoutes" },
      actions: ["insert", "find", "update", "remove"] }
  ],
  roles: []
})
